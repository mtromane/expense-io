> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mettre en place le harnais Playwright boîte noire, le filet de parité P0 (API, UI, exploitation), les tests de défauts techniques P0 en rouge → vert et la CI GitHub Actions bloquante.

**Architecture:** Les tests attaquent l'application lancée en build de production (`next start`) sur le port 3000, plus une instance « de panne » sur le port 3001 (S3 injoignable, secret de session distinct). Les acteurs (entreprises, utilisateurs, hiérarchie) sont créés par `scripts/seed-e2e.mjs` à partir de `tests/fixtures/dataset.json` ; les notes sont créées par chaque test via l'API. Les défauts techniques sont déclarés dans un registre ; tant qu'un défaut n'est pas corrigé, son test est annoncé en échec attendu (`test.fail`).

**Tech Stack:** Playwright Test (API + Chromium), Node.js 24, Next.js 16, PostgreSQL 16, LocalStack S3, GitHub Actions.

**Spec:** `docs/tests/specs/2026-10-12-strategie-tests-design.md`

## Review Focus

1. Mauvais mot de passe : aucune session ne doit être créée, même si Auth.js renvoie une redirection → test Task 3 (`mauvais mot de passe`).
2. Hiérarchie modifiée par un test puis relance du seed : la hiérarchie doit revenir à l'état du dataset → test Task 2 (`seed-e2e restaure la hiérarchie`).
3. Limite de taille du reçu exactement à 10 Mo (accepté) et 10 Mo + 1 octet (refusé) → tests Task 4.
4. Montants affichés au format français avec espace insécable avant `€` → regex `\s` dans Task 7 (E2E-03).
5. Accès concurrent de plusieurs workers aux mêmes acteurs : chaque assertion de liste porte sur des notes à description unique créées par le test, jamais sur un nombre total → Tasks 4, 5.

```js
// Crée les acteurs de test décrits dans tests/fixtures/dataset.json
// (entreprises, utilisateurs, hiérarchie) — jamais de notes : chaque test
// crée les siennes via l'API. Idempotent : chaque exécution remet la
// hiérarchie dans l'état du dataset.
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import bcrypt from 'bcryptjs'
import { config as loadEnv } from 'dotenv'
import { Pool } from 'pg'

loadEnv({ quiet: true })

const dataset = JSON.parse(
  readFileSync(new URL('../tests/fixtures/dataset.json', import.meta.url), 'utf8')
)
const pool = new Pool({ connectionString: process.env.DATABASE_URL })
// Coût bcrypt réduit : comptes de test uniquement, la vérification au login
// fonctionne quel que soit le coût.
const passwordHash = await bcrypt.hash(dataset.password, 4)

function emailFor(userKey, domain) {
  return `${userKey.toLowerCase()}@${domain}`
}

try {
  const actors = {}
  const pendingManagers = []

  for (const company of dataset.companies) {
    const companyResult = await pool.query(
      `INSERT INTO companies (name, email_domain) VALUES ($1, $2)
       ON CONFLICT (email_domain) DO UPDATE SET name = EXCLUDED.name
       RETURNING id`,
      [company.key, company.domain]
    )
    const companyId = companyResult.rows[0].id

    for (const user of company.users) {
      const email = emailFor(user.key, company.domain)
      const userResult = await pool.query(
        `INSERT INTO users (email, name, password_hash, role, company_id, manager_id)
         VALUES ($1, $2, $3, $4, $5, NULL)
         ON CONFLICT (email) DO UPDATE SET
           name = EXCLUDED.name,
           password_hash = EXCLUDED.password_hash,
           role = EXCLUDED.role,
           company_id = EXCLUDED.company_id,
           manager_id = NULL
         RETURNING id`,
        [email, user.name, passwordHash, user.role, companyId]
      )
      actors[user.key] = {
        key: user.key,
        id: userResult.rows[0].id,
        email,
        name: user.name,
        role: user.role,
        company: company.key,
      }
      if (user.manager) pendingManagers.push([user.key, user.manager])
    }
  }

  for (const [userKey, managerKey] of pendingManagers) {
    await pool.query('UPDATE users SET manager_id = $1 WHERE id = $2', [
      actors[managerKey].id,
      actors[userKey].id,
    ])
  }

  mkdirSync(new URL('../tests/.generated/', import.meta.url), { recursive: true })
  writeFileSync(
    new URL('../tests/.generated/actors.json', import.meta.url),
    JSON.stringify(actors, null, 2)
  )
  console.log(`seed-e2e : ${Object.keys(actors).length} acteurs prêts.`)
} finally {
  await pool.end()
}
```

```ts
import { request, type APIRequestContext } from '@playwright/test'
import { actor, E2E_PASSWORD, type ActorKey } from './actors'
import { BASE_URL } from './env'

export interface Credentials {
  email: string
  password: string
}

export function credentialsOf(a: { email: string }): Credentials {
  return { email: a.email, password: E2E_PASSWORD }
}

export async function anonymous(baseURL = BASE_URL): Promise<APIRequestContext> {
  return request.newContext({ baseURL })
}

/** Connexion par les endpoints Auth.js (CSRF + callback credentials). */
export async function apiLogin(creds: Credentials, baseURL = BASE_URL): Promise<APIRequestContext> {
  const ctx = await request.newContext({ baseURL })
  const { csrfToken } = (await (await ctx.get('/api/auth/csrf')).json()) as { csrfToken: string }
  const res = await ctx.post('/api/auth/callback/credentials', {
    form: { email: creds.email, password: creds.password, csrfToken, callbackUrl: `${baseURL}/expenses` },
    maxRedirects: 0,
  })
  const { cookies } = await ctx.storageState()
  if (!cookies.some((c) => c.name.endsWith('authjs.session-token'))) {
    await ctx.dispose()
    throw new Error(
      `Connexion refusée pour ${creds.email} (HTTP ${res.status()}, location=${res.headers()['location']})`
    )
  }
  return ctx
}

export async function loginAs(key: ActorKey, baseURL = BASE_URL): Promise<APIRequestContext> {
  return apiLogin(credentialsOf(actor(key)), baseURL)
}
```

```ts
import { readFileSync } from 'node:fs'
import path from 'node:path'

export interface ReceiptFile {
  name: string
  mimeType: string
  buffer: Buffer
}

const KINDS = {
  png: { file: 'receipt.png', mimeType: 'image/png' },
  jpg: { file: 'receipt.jpg', mimeType: 'image/jpeg' },
  webp: { file: 'receipt.webp', mimeType: 'image/webp' },
  gif: { file: 'receipt.gif', mimeType: 'image/gif' },
  pdf: { file: 'receipt.pdf', mimeType: 'application/pdf' },
} as const

export type ReceiptKind = keyof typeof KINDS

export function receipt(kind: ReceiptKind = 'png'): ReceiptFile {
  const { file, mimeType } = KINDS[kind]
  return {
    name: file,
    mimeType,
    buffer: readFileSync(path.join(process.cwd(), 'tests/fixtures/receipts', file)),
  }
}

/** PNG réel complété par des zéros jusqu'à la taille exacte demandée. */
export function paddedPng(totalBytes: number): ReceiptFile {
  const base = receipt('png').buffer
  const buffer = Buffer.alloc(totalBytes)
  base.copy(buffer)
  return { name: 'big.png', mimeType: 'image/png', buffer }
}
```

```ts
export interface ExpenseNoteDto {
  id: string
  userId: string
  amount: number
  expenseDate: string
  description: string
  receiptKey: string
  status: 'pending' | 'approved' | 'rejected'
  createdAt: string
}

export interface NoteInput {
  amount?: string
  expenseDate?: string
  description?: string
  /** `null` = pas de fichier envoyé. */
  receipt?: ReceiptFile | null
}

export function uniqueDescription(label: string): string {
  return `${label} ${randomUUID().slice(0, 8)}`
}

export function postNote(ctx: APIRequestContext, input: NoteInput = {}): Promise<APIResponse> {
  const multipart: Record<string, string | ReceiptFile> = {
    amount: input.amount ?? '42.50',
    expenseDate: input.expenseDate ?? '2026-03-12',
    description: input.description ?? uniqueDescription('Note'),
  }
  const file = input.receipt === undefined ? receipt('png') : input.receipt
  if (file) multipart.receipt = file
  return ctx.post('/api/expense-notes', { multipart })
}

export async function createNote(ctx: APIRequestContext, input: NoteInput = {}): Promise<ExpenseNoteDto> {
  const res = await postNote(ctx, input)
  expect(res.status(), await res.text()).toBe(201)
  return (await res.json()) as ExpenseNoteDto
}

export async function listOwnNotes(ctx: APIRequestContext): Promise<ExpenseNoteDto[]> {
  const res = await ctx.get('/api/expense-notes')
  expect(res.status()).toBe(200)
  return (await res.json()) as ExpenseNoteDto[]
```

```ts
import { expect, test, type APIRequestContext } from '@playwright/test'
import { actor } from '../support/actors'
import { anonymous, loginAs } from '../support/auth'
import { receipt } from '../support/files'
import { createNote, uniqueDescription } from '../support/notes'
import { expectDefect } from '../defects/registry'
import { AUTHZ_MATRIX, MATRIX_ACTORS, type MatrixActor, type MatrixRow } from './authz-matrix'

const contexts = new Map<MatrixActor, APIRequestContext>()

test.beforeAll(async () => {
  contexts.set('anonymous', await anonymous())
  for (const [column, key] of Object.entries(MATRIX_ACTORS)) {
    contexts.set(column as MatrixActor, await loginAs(key))
  }
})
test.afterAll(async () => {
  await Promise.all([...contexts.values()].map((c) => c.dispose()))
})

async function resolveId(row: MatrixRow): Promise<string | null> {
  switch (row.target) {
    case 'employeeNote':
      return (await createNote(contexts.get('owner')!)).id
    case 'rootManagerNote':
      return (await createNote(contexts.get('otherManager')!)).id // otherManager = rootMgrA
    case 'userEmpA2':
      return actor('empA2').id
    case 'none':
      return null
  }
}

function requestOptions(row: MatrixRow) {
  switch (row.body) {
    case 'note':
      return {
        multipart: { amount: '12.00', expenseDate: '2026-03-12', description: uniqueDescription('Matrice'), receipt: receipt() },
      }
    case 'pdfReceipt':
      return { multipart: { receipt: receipt('pdf') } }
    case 'sameManager':
      return { data: { managerId: actor('subMgrA').id } }
    case 'none':
      return {}
  }
}

for (const row of AUTHZ_MATRIX) {
  for (const [column, cell] of Object.entries(row.expected) as [MatrixActor, MatrixRow['expected'][MatrixActor]][]) {
    const { status, defect } = typeof cell === 'number' ? { status: cell, defect: undefined } : cell
    const tag = defect ? ['@defect', `@${defect}`] : ['@parity']
    test(`AUTHZ ${row.id} ${row.method} ${row.path} — ${column} → ${status}`, { tag }, async () => {
      if (defect) expectDefect(defect)
      const id = await resolveId(row)
      const url = id ? row.path.replace(':id', id) : row.path
      const res = await contexts.get(column)!.fetch(url, { method: row.method, ...requestOptions(row) })
      expect(res.status()).toBe(status)
    })
  }
}
```
